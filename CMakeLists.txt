cmake_minimum_required(VERSION 3.16)
project(GitAdventure C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(SDL2 REQUIRED)
add_executable(gitadventure src/main.c)
target_link_libraries(gitadventure PRIVATE SDL2::SDL2)
if(TARGET SDL2::SDL2main)
  target_link_libraries(gitadventure PRIVATE SDL2::SDL2main)
endif()
